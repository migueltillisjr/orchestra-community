+# Build a short CHANGE_SUMMARY from CHANGES.md without loading the full diff.
+# Usage:

+extract_fenced_block() {
+  local heading="$1"
+  awk -v h="$heading" '
+    $0 == h { in_heading=1; next }
+    in_heading && /^```/ { if (!in_fence) { in_fence=1; next } else { exit } }
+    in_heading && in_fence { print }
+  ' CHANGES.md
+}
+
+STATUS_LINES="$(extract_fenced_block "## STAGED_FILE_STATUS")"
+FILE_LINES="$(extract_fenced_block "## STAGED_FILES")"
+
+if [[ -z "$FILE_LINES" ]]; then
+  echo "No staged changes detected; nothing to summarize for release."
+  exit 0
+fi
+
+TOTAL_FILES="$(echo "$FILE_LINES" | sed '/^\s*$/d' | wc -l | tr -d ' ')"
+ADDED_COUNT="$(echo "$STATUS_LINES" | awk '$1=="A" {c++} END {print c+0}')"
+MODIFIED_COUNT="$(echo "$STATUS_LINES" | awk '$1=="M" {c++} END {print c+0}')"
+DELETED_COUNT="$(echo "$STATUS_LINES" | awk '$1=="D" {c++} END {print c+0}')"
+RENAMED_COUNT="$(echo "$STATUS_LINES" | awk '$1=="R" {c++} END {print c+0}')"

+#!/usr/bin/env bash
+
+set -euo pipefail
+
+# Deterministically generate CHANGE_SUMMARY and a CHANGES.md artifact
+# from the staged diff of the current repository.
+#
+# Usage:
+#   ./run_generate_changes.sh
+
+REPO_ROOT="$(git rev-parse --show-toplevel)"
+cd "$REPO_ROOT"
+
+# Stage the release candidate first, per release rules.
+git add -A :/
+# Exclude generated output artifact from staged summary inputs.
+git restore --staged CHANGES.md >/dev/null 2>&1 || true
+
+STATUS_LINES="$(git diff --cached --name-status)"
+FILE_LINES="$(git diff --cached --name-only)"
+
+if [[ -z "$FILE_LINES" ]]; then
+  cat > CHANGES.md <<'EOF'
+# CHANGES
+
+## CHANGE_SUMMARY
+
+Not generated by script. Derive this summary from STAGED_FILE_STATUS and STAGED_DIFF.
+
+## STAGED_FILE_STATUS
+
+```text
+(none)
+```
+
+## STAGED_FILES
+
+```text
+(none)
+```
+
+## STAGED_DIFF
+
+```diff
+(none)
+```
+EOF
+
+  echo "CHANGES.md generated. CHANGE_SUMMARY was intentionally not generated by script."
+  exit 0
+fi
+
+DIFF_CONTENT="$(git diff --cached)"
+
+{
+  cat <<'EOF'
+# CHANGES
+
+## CHANGE_SUMMARY
+
+Not generated by script. Derive this summary from STAGED_FILE_STATUS and STAGED_DIFF.
+
+EOF
+  cat <<'EOF'
+## STAGED_FILE_STATUS
+
+```text
+EOF
+  printf '%s\n' "$STATUS_LINES"
+  cat <<'EOF'
+```
+
+## STAGED_FILES
+
+```text
+EOF
+  printf '%s\n' "$FILE_LINES"
+  cat <<'EOF'
+```
+
+## STAGED_DIFF
+
+```diff
+EOF
+  printf '%s\n' "$DIFF_CONTENT"
+  cat <<'EOF'
+```
+EOF
+} > CHANGES.md
+
+# Keep CHANGES.md as generated output, not part of the staged release set by default.
+git restore --staged CHANGES.md >/dev/null 2>&1 || true
+
+echo "CHANGES.md generated."
+echo "CHANGE_SUMMARY was intentionally not generated by script."
